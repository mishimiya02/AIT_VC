const ID_HOJA = "11kxCAG0UFiTO6KodPXk9h1f31UublGACTMNRjQnx4D8";
const NOMBRE_HOJA = "relacion_vacaciones"; // ⚠️ cámbialo


function doGet(e) {
  const hoja = SpreadsheetApp.openById(ID_HOJA).getSheetByName(NOMBRE_HOJA);
  const datos = hoja.getDataRange().getValues();

  // encabezados
  const headers = datos[0];

  // índice de columna AREA (busca dinámicamente)
  const indexArea = headers.findIndex(h => h.toString().toLowerCase().includes("area"));

  let resultado = datos;

  if (e.parameter.area && indexArea !== -1) {
    resultado = datos.filter((fila, i) => {
      if (i === 0) return true;
      return fila[indexArea].toString().toLowerCase() === e.parameter.area.toLowerCase();
    });
  }

  return ContentService
    .createTextOutput(JSON.stringify(resultado))
    .setMimeType(ContentService.MimeType.JSON);
}
