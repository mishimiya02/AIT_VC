function doGet(e) {
  const action = e.parameter.action;

  if (action === "getData") {
    return sendJSON(getData());
  }

  if (action === "filter") {
    const tipo = e.parameter.tipo || "";
    const concepto = e.parameter.concepto || "";
    const estado = e.parameter.estado || "";
    return sendJSON(filtrar(tipo, concepto, estado));
  }

  if (action === "getTipos") {
    return sendJSON(getTiposContrato());
  }

  return sendJSON({error: "Acción inválida"});
}

// ---------------------------------
// Obtiene todos los datos de la hoja
// ---------------------------------
function getData() {
  const ss = SpreadsheetApp.openById(
    "1cGJk_Dg2DjT9h2_cH1ScQ5cgCxlPtg2eQXQIzPv79gk"
  );

  const sheet = ss.getSheetByName("contratos");
  const data = sheet.getDataRange().getValues();
  const rows = data.slice(1); // Ignora cabecera

  return rows.map(r => {
    const raw = r[12]; // Columna M: Fecha Fin
    
    return {
      id: r[1],               // B
      numCliente: r[2],       // C
      numDctSap: r[3],        // D
      numContrato: r[4],      // E
      tipoContrato: r[5],     // F
      concepto: r[8],         // I
      fechaFinRaw: parseDate(raw),  // para filtrar
      fechaFin: formatDate(raw)     // para mostrar
    };
  });
}

// ---------------------------------
// Determina el estado del contrato basado en la fecha
// ---------------------------------
function determinarEstado(fechaFin) {
  if (!fechaFin || !(fechaFin instanceof Date) || isNaN(fechaFin)) {
    return "sinFecha";
  }
  
  const hoy = new Date();
  hoy.setHours(0, 0, 0, 0);
  
  const fechaFinLimpia = new Date(fechaFin);
  fechaFinLimpia.setHours(0, 0, 0, 0);
  
  const diffTime = fechaFinLimpia.getTime() - hoy.getTime();
  const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
  
  if (diffDays < 0) {
    return "vencido";
  } else if (diffDays <= 30) {
    return "30dias";
  } else if (diffDays <= 60) {
    return "60dias";
  } else {
    return "vigente";
  }
}

// ---------------------------------
// Filtrado por Tipo, Concepto y Estado - CORREGIDO
// ---------------------------------
function filtrar(tipo, concepto, estado) {
  const datos = getData();

  return datos.filter(row => {
    let ok = true;

    // Filtro por tipo
    if (tipo)
      ok = ok && row.tipoContrato === tipo;

    // Filtro por concepto
    if (concepto)
      ok = ok && row.concepto.toLowerCase().includes(concepto.toLowerCase());

    // Filtro por estado
    if (estado) {
      const estadoContrato = determinarEstado(row.fechaFinRaw);
      ok = ok && estadoContrato === estado;
    }

    return ok;
  });
}

// ---------------------------------
// Obtener lista de tipos de contrato
// ---------------------------------
function getTiposContrato() {
  const datos = getData();
  const tipos = [...new Set(datos.map(r => r.tipoContrato))];
  return tipos.filter(tipo => tipo !== ""); // Filtrar vacíos
}

// ---------------------------------
// Convierte a JSON
// ---------------------------------
function sendJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// ---------------------------------
// Formatea fecha para mostrar en HTML
// ---------------------------------
function formatDate(value) {
  if (!value) return "";

  try {
    const date = new Date(value);
    const day = ("0" + date.getDate()).slice(-2);
    const month = ("0" + (date.getMonth() + 1)).slice(-2);
    const year = date.getFullYear();
    return `${day}/${month}/${year}`;
  } catch (e) {
    return value;
  }
}

// ---------------------------------
// Convierte cualquier valor a Date válido
// ---------------------------------
function parseDate(value) {
  if (!value) return null;

  // Ya es Date
  if (Object.prototype.toString.call(value) === '[object Date]' && !isNaN(value)) {
    return value;
  }

  // Número (serial de Sheets)
  if (typeof value === "number") {
    return new Date(Math.round((value - 25569) * 86400 * 1000));
  }

  // String tipo "20/02/2021" o "20-02-2021"
  if (typeof value === "string") {
    const parts = value.split(/[\/\-]/);
    if (parts.length === 3) {
      // dd/mm/yyyy
      const day = parseInt(parts[0], 10);
      const month = parseInt(parts[1], 10) - 1;
      const year = parseInt(parts[2], 10);
      return new Date(year, month, day);
    }
  }

  // Intenta parseo estándar
  const d = new Date(value);
  return isNaN(d) ? null : d;
}
